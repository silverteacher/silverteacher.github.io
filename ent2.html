    <script>
        let score = 0;
        let questionsAnswered = 0;
        const totalQuestions = 4;

        document.querySelectorAll('.option').forEach(option => {
            option.addEventListener('click', function() {
                const quizItem = this.closest('.quiz-item');
                const allOptions = quizItem.querySelectorAll('.option');
                const isCorrect = this.dataset.answer === 'correct';
                
                // Vérifier si la question a déjà été répondue
                if (quizItem.classList.contains('answered')) {
                    return;
                }
                
                // Marquer la question comme répondue
                quizItem.classList.add('answered');
                
                // Désactiver tous les options
                allOptions.forEach(opt => {
                    opt.style.pointerEvents = 'none';
                    if (opt.dataset.answer === 'correct') {
                        opt.classList.add('correct');
                    } else if (opt === this && !isCorrect) {
                        opt.classList.add('incorrect');
                    }
                });

                // Sélectionner le bon feedback
                const correctFeedback = quizItem.querySelector('.correct-feedback');
                const incorrectFeedback = quizItem.querySelector('.incorrect-feedback');
                
                // Mettre à jour le score
                if (isCorrect) {
                    score++;
                    correctFeedback.classList.add('show', 'correct');
                } else {
                    incorrectFeedback.classList.add('show', 'incorrect');
                }

                questionsAnswered++;
                
                // Mettre à jour le score et la barre de progression
                document.getElementById('score').textContent = score;
                const progress = (questionsAnswered / totalQuestions) * 100;
                document.getElementById('progress').style.width = progress + '%';

                // Animation de célébration si toutes les questions sont bonnes
                if (questionsAnswered === totalQuestions && score === totalQuestions) {
                    setTimeout(() => {
                        alert('🎉 Félicitations ! Vous avez obtenu un score parfait ! Vous maîtrisez parfaitement les applications de l\'ENT Néo !');
                    }, 1000);
                }
            });
        });

        // Animation des cartes de ressources
        const observerOptions = {
            threshold: 0.1,
            rootMargin: '0px 0px -50px 0px'
        };

        const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    entry.target.style.opacity = '1';
                    entry.target.style.transform = 'translateY(0)';
                }
            });
        }, observerOptions);
