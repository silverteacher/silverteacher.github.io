        function shuffleArray(array) {

            const newArray = [...array];

            for (let i = newArray.length - 1; i > 0; i--) {

                const j = Math.floor(Math.random() * (i + 1));

                [newArray[i], newArray[j]] = [newArray[j], newArray[i]];

            }

            return newArray;

        }

        function createConfetti() {

        function placeCard(position) {

            if (!gameState.selectedCard) return;

            

            const card = gameState.selectedCard;

            const correctPosition = findCorrectPosition(card);

            gameState.totalAttempts++;

            

            if (position === correctPosition) {

                // Placement correct

                gameState.timeline.splice(position, 0, card);

                gameState.playerHand = gameState.playerHand.filter(c => c.id !== card.id);

                gameState.score++;

                

                // Ajouter animation et confetti

                setTimeout(() => {

                    const placedCard = document.querySelector(`[data-card-id="${card.id}"]`);

                    if (placedCard) {

                        placedCard.classList.add('correct-placement');

                    }

                    createConfetti();

                }, 100);

                

                showMessage("Excellent ! Carte bien placée ! 🎉", "success");

                

                // Vérifier si le joueur a gagné

                if (gameState.playerHand.length === 0) {

                    const accuracy = Math.round((gameState.score / gameState.totalAttempts) * 100);

                    showMessage(`🏆 Félicitations ! Vous avez terminé !\nScore: ${gameState.score}/4\nPrécision: ${accuracy}%`, "success");

                    gameState.gameStarted = false;

                    document.getElementById('reveal-btn').disabled = true;

                    createConfetti();

                    // Confetti supplémentaire pour la victoire

                    setTimeout(() => createConfetti(), 1000);

                    setTimeout(() => createConfetti(), 2000);

                }

            } else {

                // Placement incorrect - on montre juste l'erreur, pas de nouvelle carte

                showMessage(`❌ Incorrect ! Cette invention date de ${card.year}.\nEssayez de la replacer !`, "error-msg");

                    cardEl.classList.add('error');

                    setTimeout(() => cardEl.classList.remove('error'), 500);

                }

            }

            

            gameState.selectedCard = null;

            document.getElementById('placement-buttons').style.display = 'none';

            updateDisplay();

        }


        function findCorrectPosition(card) {

            for (let i = 0; i <= gameState.timeline.length; i++) {

                if (i === 0 && card.year <= gameState.timeline[0].year) {

                    return 0;

                } else if (i === gameState.timeline.length && card.year >= gameState.timeline[i-1].year) {

                    return i;

                } else if (i > 0 && i < gameState.timeline.length && 

                          card.year >= gameState.timeline[i-1].year && 

                          card.year <= gameState.timeline[i].year) {

                    return i;

                }

            }

            return gameState.timeline.length;

        }


        function revealSolution() {

            if (!gameState.gameStarted || gameState.playerHand.length === 0) return;

            

            // Trier toutes les cartes par année

            const allCardsSorted = [...gameState.allCards].sort((a, b) => a.year - b.year);

            

            let solutionText = "🔍 SOLUTION COMPLÈTE :\n\n";

            allCardsSorted.forEach((card, index) => {

                const isPlaced = gameState.timeline.some(c => c.id === card.id);

                const isInHand = gameState.playerHand.some(c => c.id === card.id);

                const status = isPlaced ? "✅" : (isInHand ? "🔄" : "");

                solutionText += `${index + 1}. ${card.year} - ${card.event} ${status}\n`;

            });

            

            solutionText += "\n✅ = Déjà placé\n🔄 = Dans votre main";

            

            showMessage(solutionText, "info");

        }


        function updateDisplay() {

            updateTimelineDisplay();

            updateHandDisplay();

            updateGameInfo();

        }


        function updateTimelineDisplay() {

            const timeline = document.getElementById('timeline');

            timeline.innerHTML = '';

            

            if (gameState.timeline.length === 0) {

                timeline.innerHTML = '<p style="color: #666; text-align: center; width: 100%;">La timeline apparaîtra ici</p>';

                return;

            }

            

            gameState.timeline.forEach(card => {

                const cardEl = createCardElement(card, true, false);

                cardEl.classList.add('placed');

                timeline.appendChild(cardEl);

            });

        }


        function updateHandDisplay() {

            const handCards = document.getElementById('hand-cards');

            handCards.innerHTML = '';

            

            gameState.playerHand.forEach(card => {

                const cardEl = createCardElement(card, false, true);

                if (gameState.selectedCard && gameState.selectedCard.id === card.id) {

                    cardEl.classList.add('selected');

                }

                handCards.appendChild(cardEl);

            });

        }


        function updateGameInfo() {

            document.getElementById('cards-remaining').textContent = gameState.playerHand.length;

            document.getElementById('score').textContent = `${gameState.score}/4`;

            const accuracy = gameState.totalAttempts > 0 ? Math.round((gameState.score / gameState.totalAttempts) * 100) : 0;

            document.getElementById('accuracy').textContent = `${accuracy}%`;

        }


        function showMessage(message, type) {

            const messageEl = document.getElementById('game-message');

            messageEl.textContent = message;

            messageEl.className = `game-message ${type}`;

            messageEl.style.display = 'block';

            

            // Pour les règles et solution, on garde le message plus longtemps

            const timeout = (type === 'info' && (message.includes('RÈGLES') || message.includes('SOLUTION'))) ? 10000 : 4000;

            

            setTimeout(() => {

                messageEl.style.display = 'none';

            }, timeout);

        }
