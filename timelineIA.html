  <p class="instructions">Replace les événements dans l’ordre chronologique. Tu peux ajouter des événements ou afficher la solution.</p>

  <script>

    let allEvents = [

      { year: 1950, text: "Turing publie son article sur l’intelligence des machines" },

      { year: 1956, text: "Conférence de Dartmouth : naissance officielle de l’IA" },

      { year: 1966, text: "Le chatbot ELIZA simule un psy" },

      { year: 1980, text: "Les systèmes experts gagnent du terrain en entreprise" },

      { year: 1997, text: "Deep Blue bat Kasparov aux échecs" },

      { year: 2006, text: "Naissance du deep learning moderne (Geoffrey Hinton)" },

      { year: 2012, text: "AlexNet pulvérise ImageNet avec un réseau de neurones profonds" },

      { year: 2016, text: "AlphaGo bat Lee Sedol au jeu de Go" },

      { year: 2022, text: "Lancement public de ChatGPT" },

      { year: 2023, text: "Explosion des générateurs d'images IA comme Midjourney" }

    ];


    const cardContainer = document.getElementById('cardContainer');

    const timeline = document.getElementById('timeline');

    const message = document.getElementById('message');

    const profForm = document.getElementById('profForm');


    let placedEvents = [];


    function createCard(event) {

      const card = document.createElement('div');

      card.className = 'card';

      card.draggable = true;

      card.textContent = event.text;

      card.dataset.year = event.year;


      card.addEventListener('dragstart', () => card.classList.add('dragging'));

      card.addEventListener('dragend', () => card.classList.remove('dragging'));


      cardContainer.appendChild(card);

    }


    function shuffleAndDisplay(events) {

      const shuffled = [...events].sort(() => Math.random() - 0.5);

      shuffled.forEach(createCard);

    }


    shuffleAndDisplay(allEvents);


    timeline.addEventListener('dragover', e => e.preventDefault());


    timeline.addEventListener('drop', e => {

      e.preventDefault();

      const draggingCard = document.querySelector('.dragging');

      if (!draggingCard) return;


      const year = parseInt(draggingCard.dataset.year, 10);

      const isCorrect = placedEvents.length === 0 || placedEvents.every((ev, i) => {

        return i === placedEvents.length - 1 ? ev.year < year : true;

      });


      if (isCorrect) {

        draggingCard.classList.add('placed');

        draggingCard.draggable = false;

        timeline.appendChild(draggingCard);

        placedEvents.push({ year, text: draggingCard.textContent });

        message.textContent = '';

      } else {

        draggingCard.classList.add('error');

        setTimeout(() => draggingCard.classList.remove('error'), 500);

        message.textContent = "Mauvais ordre ! Essaie encore.";

      }

    });


    function toggleProf() {

      profForm.style.display = profForm.style.display === 'none' ? 'flex' : 'none';

    }


    function addEvent() {

      const year = parseInt(document.getElementById('newYear').value.trim(), 10);

      const text = document.getElementById('newText').value.trim();

      if (!year || !text) {

        alert("Remplis les deux champs.");

        return;

      }

      allEvents.push({ year, text });

      createCard({ year, text });

      document.getElementById('newYear').value = '';

      document.getElementById('newText').value = '';

    }


    function showSolution() {

      // Vider la timeline et y placer les événements triés

      timeline.innerHTML = '';

      const sorted = [...allEvents].sort((a, b) => a.year - b.year);

      sorted.forEach(ev => {

        const card = document.createElement('div');

        card.className = 'card placed';

        card.textContent = `${ev.year} – ${ev.text}`;

        timeline.appendChild(card);

      });

      message.textContent = "Voici l'ordre chronologique correct.";

    }
