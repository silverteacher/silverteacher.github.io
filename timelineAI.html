        function shuffleArray(array) {

            var newArray = array.slice();

            for (var i = newArray.length - 1; i > 0; i--) {

                var j = Math.floor(Math.random() * (i + 1));

                var temp = newArray[i];

                newArray[i] = newArray[j];

                newArray[j] = temp;

            }

            return newArray;

        }

        function placeCard(position) {

            if (!gameState.selectedCard) return;

            var correctPosition = findCorrectPosition(card);

            gameState.totalAttempts++;

            

            if (position === correctPosition) {

                gameState.timeline.splice(position, 0, card);

                gameState.playerHand = gameState.playerHand.filter(function(c) { return c.id !== card.id; });

                gameState.score++;

                    var placedCard = document.querySelector('[data-card-id="' + card.id + '"]');

                    gameState.gameStarted = false;

                    document.getElementById('reveal-btn').disabled = true;

                    createConfetti();

            document.getElementById('placement-buttons').style.display = 'none';

            updateDisplay();

        }

                } else if (i === gameState.timeline.length && card.year >= gameState.timeline[i-1].year) {

                    return i;

                } else if (i > 0 && i < gameState.timeline.length && 

                          card.year >= gameState.timeline[i-1].year && 

                          card.year <= gameState.timeline[i].year) {

                    return i;

                }

            }

            return gameState.timeline.length;

        }


        function revealSolution() {

            if (!gameState.gameStarted || gameState.playerHand.length === 0) return;

            

            var allCardsSorted = gameState.allCards.slice().sort(function(a, b) { return a.year - b.year; });

            

            var solutionText = "🔍 SOLUTION COMPLÈTE :\n\n";

            for (var i = 0; i < allCardsSorted.length; i++) {

                var card = allCardsSorted[i];

                var isPlaced = gameState.timeline.some(function(c) { return c.id === card.id; });

                var isInHand = gameState.playerHand.some(function(c) { return c.id === card.id; });

                var status = isPlaced ? "✅" : (isInHand ? "🔄" : "");

                solutionText += (i + 1) + ". " + card.year + " - " + card.event + " " + status + "\n";

            }

            

            solutionText += "\n✅ = Déjà placé\n🔄 = Dans votre main";

            

            showMessage(solutionText, "info");

        }

        function updateTimelineDisplay() {

            var timeline = document.getElementById('timeline');

            timeline.innerHTML = '';

            

            if (gameState.timeline.length === 0) {

                timeline.innerHTML = '<p style="color: #666; text-align: center; width: 100%;">La timeline apparaîtra ici</p>';

                return;

            }

            

            for (var i = 0; i < gameState.timeline.length; i++) {

                var card = gameState.timeline[i];

                var cardEl = createCardElement(card, true, false);

                cardEl.classList.add('placed');

                timeline.appendChild(cardEl);

            }

        }


        function updateHandDisplay() {

            var handCards = document.getElementById('hand-cards');

            handCards.innerHTML = '';

            

            for (var i = 0; i < gameState.playerHand.length; i++) {

                var card = gameState.playerHand[i];

                var cardEl = createCardElement(card, false, true);

                if (gameState.selectedCard && gameState.selectedCard.id === card.id) {

                    cardEl.classList.add('selected');

                }

                handCards.appendChild(cardEl);

            }

        }


        function updateGameInfo() {

            document.getElementById('accuracy').textContent = accuracy + "%";

        }


        function showMessage(message, type) {

        }
