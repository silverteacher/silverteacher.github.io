        function switchTab(activeTabId) {

        function handleTabKeydown(e) {

            const tabButtons = Array.from(document.querySelectorAll('.tab-button'));

            const currentIndex = tabButtons.indexOf(e.target);

            

            switch (e.key) {

                case 'ArrowLeft':

                    e.preventDefault();

                    const prevIndex = currentIndex > 0 ? currentIndex - 1 : tabButtons.length - 1;

                    tabButtons[prevIndex].focus();

                    switchTab(tabButtons[prevIndex].id);

                    break;

                case 'ArrowRight':

                    e.preventDefault();

                    const nextIndex = currentIndex < tabButtons.length - 1 ? currentIndex + 1 : 0;

                    tabButtons[nextIndex].focus();

                    switchTab(tabButtons[nextIndex].id);

                    break;

                case 'Home':

                    e.preventDefault();

                    tabButtons[0].focus();

                    switchTab(tabButtons[0].id);

                    break;

                case 'End':

                    e.preventDefault();

                    tabButtons[tabButtons.length - 1].focus();

                    switchTab(tabButtons[tabButtons.length - 1].id);

                    break;

            }

        }

        function handleSequenceDragOver(e) {

            e.preventDefault();

            e.dataTransfer.dropEffect = 'move';

            

            if (this === draggedElement) return;

            

            const rect = this.getBoundingClientRect();

            const midpoint = rect.left + rect.width / 2;

            const mouseX = e.clientX;

            

            // Supprimer les anciens indicateurs

            document.querySelectorAll('.drop-indicator').forEach(el => el.remove());

            

            // Créer un nouvel indicateur

            const indicator = document.createElement('div');

            indicator.className = 'drop-indicator absolute top-0 bottom-0 w-1 bg-blue-500 z-10 rounded';

            indicator.style.pointerEvents = 'none';

            

            if (mouseX < midpoint) {

                // Insérer avant

                indicator.style.left = '-2px';

                this.style.position = 'relative';

                this.appendChild(indicator);

            } else {

                // Insérer après

                indicator.style.right = '-2px';

                this.style.position = 'relative';

                this.appendChild(indicator);

            }

        }


        async function handleSequenceDrop(e) {

            e.preventDefault();

            e.stopPropagation();

            

            this.classList.remove('drag-over-sequence');

            document.querySelectorAll('.drop-indicator').forEach(el => el.remove());

            

            if (this === draggedElement) return;

            

            try {

                const dragData = JSON.parse(e.dataTransfer.getData('text/plain'));

                

                if (dragData.isReordering) {

                    await reorderSequenceItems(dragData.itemId, this.dataset.id, e);

                }

            } catch (error) {

                console.error('Erreur lors du réarrangement:', error);

                showToast('Erreur lors du réarrangement', 'error');

            }

        }


        async function reorderSequenceItems(draggedItemId, targetItemId, event) {

            const draggedItem = items.find(i => i.__backendId === draggedItemId);

            const targetItem = items.find(i => i.__backendId === targetItemId);

            

            if (!draggedItem || !targetItem) return;

            

            const sequenceItems = items.filter(item => item.position >= 1000 && item.position < 2000)

                                      .sort((a, b) => a.position - b.position);

            

            const draggedIndex = sequenceItems.findIndex(i => i.__backendId === draggedItemId);

            const targetIndex = sequenceItems.findIndex(i => i.__backendId === targetItemId);

            

            if (draggedIndex === -1 || targetIndex === -1) return;

            

            // Déterminer si on insère avant ou après

            const rect = event.target.getBoundingClientRect();

            const midpoint = rect.left + rect.width / 2;

            const mouseX = event.clientX;

            const insertBefore = mouseX < midpoint;

            

            // Calculer la nouvelle position

            let newIndex = targetIndex;

            if (!insertBefore && draggedIndex < targetIndex) {

                // Pas de changement d'index si on insère après et qu'on vient de la gauche

            } else if (insertBefore && draggedIndex > targetIndex) {

                // Pas de changement d'index si on insère avant et qu'on vient de la droite

            } else if (!insertBefore) {

                newIndex = targetIndex + 1;

            }

            

            // Réorganiser le tableau

            const newSequence = [...sequenceItems];

            const [movedItem] = newSequence.splice(draggedIndex, 1);

            newSequence.splice(newIndex > draggedIndex ? newIndex - 1 : newIndex, 0, movedItem);

            

            // Mettre à jour les positions

            const updatePromises = newSequence.map((item, index) => {

                item.position = 1000 + index;

                return window.dataSdk.update(item);

            });

            

            try {

                await Promise.all(updatePromises);

                showToast('Ordre mis à jour ✅', 'success');

            } catch (error) {

                console.error('Erreur lors de la mise à jour des positions:', error);

                showToast('Erreur lors du réarrangement', 'error');

            }

        }

        async function handleDrop(e) {

            e.preventDefault();

            this.classList.remove('drag-over');

            

            if (!draggedElement) return;

            

            try {

                const dragData = JSON.parse(e.dataTransfer.getData('text/plain'));

                

                if (dragData.isNewCard) {

                    await handleNewCardDrop(dragData, this.id);

                } else {

                    await handleExistingItemDrop(dragData, this.id);

                }

            } catch (error) {

                console.error('Erreur lors du drop:', error);

                showToast('Erreur lors du déplacement', 'error');

            }

        }


        async function handleNewCardDrop(dragData, dropZoneId) {

            if (dropZoneId !== 'chemin-fer') return;

            

            if (items.length >= 999) {

                showToast('Limite atteinte : maximum 999 éléments', 'error');

                return;

            }

            

            const cardData = dragData.type === 'tool' ? 

                availableTools.find(t => t.id === dragData.cardId) :

                availableAnimations.find(a => a.id === dragData.cardId);

            

            if (!cardData) return;

            

            const position = 1000 + items.filter(i => i.position >= 1000 && i.position < 2000).length;

            

            // Générer un ID unique pour permettre la sélection infinie

            const uniqueId = `${cardData.id}-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;

            

            const newItem = {

                id: uniqueId,

                type: dragData.type,

                name: cardData.name,

                icon: cardData.icon,

                description: cardData.description,

                position: position,

                notes: '',

                webinar_type: '',

                participants: 0,

                validated: false,

                created_at: new Date().toISOString()

            };

            

            const result = await window.dataSdk.create(newItem);

            if (result.isOk) {

                showToast(`${cardData.name} ajouté au chemin de fer`, 'success');

            } else {

                showToast('Erreur lors de l\'ajout', 'error');

            }

        }


        async function handleExistingItemDrop(dragData, dropZoneId) {

            const item = items.find(i => i.__backendId === dragData.itemId);

            if (!item) return;

            

            if (dropZoneId === 'chemin-fer') {

                const newPosition = 1000 + items.filter(i => i.position >= 1000 && i.position < 2000).length;

                item.position = newPosition;

                const result = await window.dataSdk.update(item);

                if (!result.isOk) {

                    showToast('Erreur lors du déplacement', 'error');

                }

            }

        }

        function wrapText(ctx, text, maxWidth) {

            const words = text.split(' ');

            const lines = [];

            let currentLine = words[0];

            

            for (let i = 1; i < words.length; i++) {

                const word = words[i];

                const width = ctx.measureText(currentLine + ' ' + word).width;

                if (width < maxWidth) {

                    currentLine += ' ' + word;

                } else {

                    lines.push(currentLine);

                    currentLine = word;

                }

            }

            lines.push(currentLine);

            return lines;

        }
